# These docs are archived and every page carries `<meta name="robots"
# content="noindex">`. Search crawlers are deliberately NOT disallowed here:
# they have to be able to fetch a page to see that tag. Adding `Disallow: /`
# for them would leave already-indexed URLs in search results forever.
#
# The rules below only block AI training crawlers, which is a separate concern.

# Block OpenAI's crawler
User-agent: GPTBot
Disallow: /

# Block Anthropic's crawler
User-agent: ClaudeBot
Disallow: /

# Block Google's AI crawler
User-agent: Google-Extended
Disallow: /

# Block Perplexity
User-agent: PerplexityBot
Disallow: /

# Block Meta/Facebook AI
User-agent: FacebookBot
Disallow: /

# Block other common AI crawlers
User-agent: Omgilibot
Disallow: /
User-agent: Omgili
Disallow: /
User-agent: CCBot
Disallow: /
User-agent: ChatGPT-User
Disallow: /
